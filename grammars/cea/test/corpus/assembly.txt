{ Example script }
{$STRICT}
define(address,Game:Update+10)
define(bytes,F3 0F 10 45 D8)

[ENABLE]
assert(address,bytes)
alloc(newmem,$1000,address)
label(code)
label(return)

newmem:
  movss xmm0,[rbp-28]
  addsd xmm0,xmm1 // add the new value
  jmp return

address:
  jmp newmem
  nop
return:

[DISABLE]
address:
  db bytes
dealloc(newmem)
[module.pointer]+offset:
  db 00
